Rust Worker
data transform
02
use iii_sdk::{
    register_worker, IIIError,
    Value,
};
use serde_json::json;

async fn transform(
    input: Value,
) -> Result<Value, IIIError> {
    let nums: Vec<f64> =
        serde_json::from_value(input)?;
    let result: Vec<f64> =
        nums.iter().map(|x| x * 2.0).collect();
    Ok(json!(result))
}

#[tokio::main]
async fn main() -> Result<(), IIIError> {
    let iii = register_worker(
        "ws://localhost:49134",
        InitOptions::default(),
    )?;
    iii.register_function(
        "data::transform",
        transform,
    );
    Ok(())
}
